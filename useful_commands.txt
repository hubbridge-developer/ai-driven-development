# deleting repos additionally needs "delete_repo" scope.
# hubbridge-developer is a USER account — use /user/repos, not /orgs/... endpoints.

# =====================
# GitHub — spec repo management (hubbridge-developer/xspec-specs)
# =====================

# Create the spec repo under the user account
GITHUB_PAT=$(grep '^GITHUB_PAT=' /Users/ozgursedefoglu/Projects/ai-driven-development/.env | head -1 | cut -d= -f2) && curl -s -X POST -H "Authorization: Bearer $GITHUB_PAT" -H "Accept: application/vnd.github+json" "https://api.github.com/user/repos" -d '{"name":"xspec-specs","description":"ADD specification repository","private":false}' | python3 -c "import sys,json; d=json.load(sys.stdin); print(f'{d.get(\"full_name\",\"\")} - {d.get(\"html_url\",d.get(\"message\",\"\"))}')"

# Delete the spec repo (PAT needs delete_repo scope; expect 204)
GITHUB_PAT=$(grep '^GITHUB_PAT=' /Users/ozgursedefoglu/Projects/ai-driven-development/.env | head -1 | cut -d= -f2) && curl -s -X DELETE -H "Authorization: Bearer $GITHUB_PAT" -H "Accept: application/vnd.github+json" "https://api.github.com/repos/hubbridge-developer/xspec-specs" -w "\n%{http_code}"

# List all branches
GITHUB_PAT=$(grep '^GITHUB_PAT=' /Users/ozgursedefoglu/Projects/ai-driven-development/.env | head -1 | cut -d= -f2) && curl -s -H "Authorization: Bearer $GITHUB_PAT" -H "Accept: application/vnd.github+json" "https://api.github.com/repos/hubbridge-developer/xspec-specs/branches?per_page=100" | python3 -c "import sys,json; [print(b['name']) for b in json.load(sys.stdin)]"

# Delete ALL spec/* branches
GITHUB_PAT=$(grep '^GITHUB_PAT=' /Users/ozgursedefoglu/Projects/ai-driven-development/.env | head -1 | cut -d= -f2) && for BRANCH in $(curl -s -H "Authorization: Bearer $GITHUB_PAT" -H "Accept: application/vnd.github+json" "https://api.github.com/repos/hubbridge-developer/xspec-specs/branches?per_page=100" | python3 -c "import sys,json; [print(b['name']) for b in json.load(sys.stdin) if b['name'].startswith('spec/')]"); do curl -s -o /dev/null -w "Deleted $BRANCH (%{http_code})\n" -X DELETE -H "Authorization: Bearer $GITHUB_PAT" -H "Accept: application/vnd.github+json" "https://api.github.com/repos/hubbridge-developer/xspec-specs/git/refs/heads/$BRANCH"; done

# List open PRs
GITHUB_PAT=$(grep '^GITHUB_PAT=' /Users/ozgursedefoglu/Projects/ai-driven-development/.env | head -1 | cut -d= -f2) && curl -s -H "Authorization: Bearer $GITHUB_PAT" -H "Accept: application/vnd.github+json" "https://api.github.com/repos/hubbridge-developer/xspec-specs/pulls?state=open&per_page=100" | python3 -c "import sys,json; [print(f'{p[\"number\"]} {p[\"title\"]}') for p in json.load(sys.stdin)]"

# List closed PRs
GITHUB_PAT=$(grep '^GITHUB_PAT=' /Users/ozgursedefoglu/Projects/ai-driven-development/.env | head -1 | cut -d= -f2) && curl -s -H "Authorization: Bearer $GITHUB_PAT" -H "Accept: application/vnd.github+json" "https://api.github.com/repos/hubbridge-developer/xspec-specs/pulls?state=closed&per_page=100" | python3 -c "import sys,json; [print(f'{p[\"number\"]} {p[\"title\"]}') for p in json.load(sys.stdin)]"

# Close ALL open PRs
GITHUB_PAT=$(grep '^GITHUB_PAT=' /Users/ozgursedefoglu/Projects/ai-driven-development/.env | head -1 | cut -d= -f2) && for PR in $(curl -s -H "Authorization: Bearer $GITHUB_PAT" -H "Accept: application/vnd.github+json" "https://api.github.com/repos/hubbridge-developer/xspec-specs/pulls?state=open&per_page=100" | python3 -c "import sys,json; [print(p['number']) for p in json.load(sys.stdin)]"); do curl -s -X PATCH -H "Authorization: Bearer $GITHUB_PAT" -H "Accept: application/vnd.github+json" "https://api.github.com/repos/hubbridge-developer/xspec-specs/pulls/$PR" -d '{"state":"closed"}' | python3 -c "import sys,json; d=json.load(sys.stdin); print(f'PR #{d[\"number\"]}: {d[\"state\"]}')"; done
